\section{Introduction}
\begin{frame}{Neural Model Predictive Control (NMPC)}
    In this work we have tried to combine Model Predictive Control along with PiNNs to trajectory tracking problem applied to quadrotors. We have used the Unicycle robot as simple test case for our experiments.
\end{frame}
\section{Quadrotor Model}

\begin{frame}{Quadrotor Standard Model}
    \begin{columns}[T] % Align columns at the top
        % Left Column: Text
        \begin{column}{0.45\textwidth}
            \vspace{0.4cm}
            Starting from the Translational \eqref{eq:tra_dyn} and Rotational Dynamics \eqref{eq:rot_dyn}, we have derived the simplified second order of the quadrotor.
            \begin{align}
                \label{eq:tra_dyn}
                m \ddot{p} = f_{t} + f_{g} + f_{a} + f_{d} \\
                \label{eq:rot_dyn}
                J \ \dot{\omega} + \omega \times (J \ \omega) = \tau 
            \end{align}
        \end{column}
        
        % Right Column: Equation
        \begin{column}{0.50\textwidth}
            \vspace{-0.2cm}
            \begin{align}
                \begin{cases}
                    \ddot{x} = -\Bigl(\cos\psi\, \sin\theta\, \cos\phi + \sin\psi\, \sin\phi\Bigr)\frac{T}{m}, \\
                    \ddot{y} = -\Bigl(\sin\psi\, \sin\theta\, \cos\phi - \cos\psi\, \sin\phi\Bigr)\frac{T}{m}, \\
                    \ddot{z} = -\Bigl(\cos\theta\, \cos\phi\Bigr)\dfrac{T}{m} + g, \\[1ex]
                    \ddot{\phi}   = \frac{\tau_{\phi}}{I_{x}} \\
                    \ddot{\theta} = \frac{\tau_{\theta}}{I_{y}} \\
                    \ddot{\psi}   = \frac{\tau_{\psi}}{I_{z}} 
                \end{cases}
            \end{align}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{MPC Structure}
    We have used for both robots the following structure for the MPC:
    \begin{align}
        \min_{u(0), u(1), \dots, u(N-1)} \quad & \sum_{k=0}^{N-1} \ell(x(k), u(k)) + \ell_f(x(N)) \\
        \text{s.t.} \quad & x(k+1) = f(x(k), u(k)), \quad k = 0, \dots, N-1, \\
        & x(k) \in \mathcal{X}, \quad u(k) \in \mathcal{U}, \quad k = 0, \dots, N-1, \\
        & x(0) = x_{\text{0}.}
    \end{align}
\end{frame}


\begin{frame}{MPC Cost Function}
    % MPC Cost Function Block
    \begin{block}{Cost Function}
      \centering
      \vspace{-1em} % Adjust this value as needed to reduce extra vertical space
      \begin{align*}
        \ell(x(k), u(k)) &= \sum_{k=0}^{N-1} \Bigl[  \sum_{x \in \mathcal{X}} W_{x(k)} \Bigl( x(k)_{\text{ref}} - x(k) \Bigr)^2 
        + \sum_{u \in \mathcal{U}} W_u \, u^2 \Bigr], \\
        \ell_{f}(x(N))      &= \sum_{x \in \mathcal{X}} W_{x(N)} \Bigl( x(N)_{\text{ref}} - x(N) \Bigr)^2 .
      \end{align*}
    \end{block}
    
    \vspace{-1em} % Adjust vertical spacing between blocks
    
  \end{frame}

\section{Dataset}
\begin{frame}{Structure}
    The data the we used to train our neural models come from simulation and
    some real world experiments. 
    \vspace{1em}
    \begin{itemize}
        \item Generally speaking, we can describe each our dataset as a tuple in the form $\left\{[x_{t}, u_{t}], [x_{t+1}]\right\}$, were $x_{t}$ is the state of the system at time $t$, $u_{t}$ is the control input at time $t$ and $x_{t+1}$ is the state of the system at time $t+1$.
        \item For the baseline (\textit{simple-neural-network}), no constraints on the dataset form are required, while on the physics informed models the input needs to be \textbf{constant} in a sub-interval $[0, T]$.
    \end{itemize}
\end{frame}

\begin{frame}{Problems}
    When extracting data from the real-world experiments (or the gazebo simulations), state and action \textbf{publishing rates} were a crucial aspect to take into account. 
    \vspace{0.6em}
    \begin{itemize}
        \item The action publishing rate should be high enough to make the controller safe enough
        \item The state publishing rate should be high, but it couldn't be too high, for communication latency reasons, which would have made the use of the real crazyflie impossible
    \end{itemize}
    \vspace{0.6em}
    These problems could be \textbf{partially mitigated} by using the gazebo simulator and setting ad-hoc publishing rates, or forcing the input to be constant (introducing a noisy approximation) for a certain amount of time in the real-world data.
\end{frame}
\section{Standard Neural Network}
\begin{frame}{Training Scheme}
    \begin{columns}[c]
      % Left Column: Text Description
      \begin{column}{0.55\textwidth}
        The multi-step training procedure is defined as follows:
        \vspace{1em}
        \begin{itemize}
          \item Give in input the pair $(s_t, a_t)$ to the network
          \item Perform the integration of the output in the sample time $dt$
          \item Add what we have to the input to obtain the new state $s_{t+1}$
          \item Compute the \textbf{MSE} loss
          \item Use the new state as the next input of the network
        \end{itemize}
      \end{column}
      
      % Right Column: Image
      \begin{column}{0.45\textwidth}
        \centering
        \includegraphics[width=\textwidth]{../report/assets/mlp_1.jpg} % Replace with your image file
      \end{column}
    \end{columns}
  \end{frame}

\begin{frame}{Unicycle Network}

    The neural network used to model the kinematic model of the unicycle is a simple MLP architecture with two hidden layers of 128 neurons.
    The input is given by the concatenation of the current state and the current control inputs:
    \begin{equation}
        X = [x, y, \theta, v, \omega]
    \end{equation}
    while the output represents the dynamics of the system:
    \begin{equation}
        Y = [\dot{x}, \dot{y}, \dot{\theta}]
    \end{equation}
\end{frame}

\begin{frame}{Quadrotor Network}
    The quadrotor dynamics presents a more complex model, with a state $x \in R^9$ and control inputs $u \in R^4$. For approximating such dynamics with the defined training scheme, a larger network has been tested: we have increased the number of hidden layers up to 3 and the latent dimension from 128 to 256. However, the main problem during training is related to the generation of the data.
\begin{align}
    X &= \begin{bmatrix}x & y & z & v_x & v_y & v_z & \phi & \theta & \psi \ |\ U \end{bmatrix}^T \\
    Y &= \begin{bmatrix} \dot{x} & \dot{y} & \dot{z} & \dot{v_x} & \dot{v_y} & \dot{v_z} & \dot{\phi} & \dot{\theta} & \dot{\psi} \end{bmatrix}^T
\end{align}
\end{frame}

\section{Physical Informed Neural Networks}
\begin{frame}{PiNNs}
    \begin{columns}[T]
        % Right Column: Text and Equations
        \begin{column}{0.55\textwidth}
            Physical Informed Neural Networks (PiNNs) propose an alternative approach where the solution is approximated by a neural network that is trained not only on data but also by enforcing the underlying physical laws via the differential equations.
            \vspace{1.5em}
            \begin{align}
                \mathcal{L}(\theta) = \mathcal{L}_{\text{data}}(\theta) + \mathcal{L}_{\text{phy}}(\theta)
            \end{align}
        \end{column}

        \begin{column}{0.4\textwidth}
            \centering
            \includegraphics[width=\textwidth]{../report/assets/oscillator.jpg} % Replace with your image file
            \vspace{1ex}
            \footnotesize{PiNN vs Classical MLP}
            \\
            \textbf{Problems and Limitations}:
            \begin{itemize}
                \item Cannot handle control input 
                \item Fixed initial condition
            \end{itemize}
        \end{column}
        
    \end{columns}
\end{frame}
\begin{frame}{PiNN with Controls}
    \begin{columns}[T]
        \begin{column}{0.45\textwidth}
            The Pinn-C modifies the formulation of the pinn, by dividing the evolution into sub-transitions with constant input. The network takes as input $(x(0), \bar{u}, t)$.
        \end{column}
        \begin{column}{0.5\textwidth}
            \vspace{-1em}
            \includegraphics[width=\textwidth]{../report/assets/pinc_time.png}
        \end{column}
    \end{columns}
    \textbf{Pros}:
            \begin{itemize}
                \item it can handle larger time-steps with respect to the original formulation
                \item can accept arbitrary starting conditions
            \end{itemize}
            \textbf{Cons}:
            \begin{itemize}
                \item relies heavily on the dataset 
            \end{itemize}
\end{frame}
\begin{frame}[fragile]{Training Scheme}
    \textbf{Training Loop: Loss Calculation}
    \begin{algorithmic}[1]
        \State \textbf{Input:} \( X_{\text{data}}, y \)
        \State \textbf{Initialize:} \( L \gets 0 \)
        \State \textbf{Compute Predictions:} \( \hat{y} \gets \mathcal{M}(X_{\text{data}}) \)
        \State \textbf{Update Loss with MSE:} \( L \gets L + \frac{1}{n} \sum_{i=1}^{n} (y_i - \hat{y}_i)^2 \)
        \State \textbf{Sample Input Data:} \( X \sim \mathcal{D} \)
        \State \textbf{Calculate Gradient:} \( J \gets \nabla_X \theta \)
        \State \textbf{Compute Analytical ODE:} \( \text{ode}(X) = f(X) \)
        \State \textbf{Update Loss with ODE Term:} \( L \gets L + \frac{1}{n} \sum_{i=1}^{n} \Bigl(\text{ode}(X) - \nabla_X \theta\Bigr)^2 \)
        \State \textbf{Sample a Point:} \( X_{\text{point}} \sim \mathcal{D} \)
        \State \textbf{Set to zero the Input:} \( X_{\text{input}} \gets 0 \)
        \State \textbf{Update Loss with Boundary Term:} \( L \gets L + \frac{1}{n} \sum_{i=1}^{n} (X_i - \hat{y}_i)^2 \)
        \State \textbf{Output:} Updated loss \( L \)
    \end{algorithmic}
    \end{frame}

\begin{frame}{Future Works}
    Despite all the challenges that we have encountered, the results underscore the potential of using physics-informed models along with Model Predictive Control. We have some ideas to improve the performances and solve some of the problems:
    \vspace{1em}
    \begin{itemize}
        \item \textbf{Reservoir computing} can be used to deal with unbalance and non-homogeneous input signal, while the output layers can be tuned in a physics informed fashion
        \item \textbf{Double stage training} mechanism could be used for linearly connect multiple models
        \item  \textbf{Moving average} technique in the training phase could be beneficial for the convergence of the network.
    \end{itemize}
\end{frame}
